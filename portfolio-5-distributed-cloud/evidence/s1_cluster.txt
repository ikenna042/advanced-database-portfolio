== Rows per worker for ledger_entries ==

[coordinator] SELECT nodename, count(*) AS shards, sum(rows) AS ledger_entries
       FROM (SELECT shardid, nodename, (result::bigint) AS rows
               FROM run_command_on_shards('wallet.ledger_entries', 'SELECT count(*) FROM %s') r
               JOIN citus_shards USING (shardid)) s
      GROUP BY nodename ORDER BY nodename

== Uniqueness that does not include the shard key cannot be enforced ==

[coordinator] ALTER TABLE wallet.customers ADD CONSTRAINT customers_phone_unique UNIQUE (phone)
    ERROR:  cannot create constraint on "customers"
    DETAIL:  Distributed relations cannot have UNIQUE, EXCLUDE, or PRIMARY KEY constraints that do not include the partition column (with an equality operator if EXCLUDE).
